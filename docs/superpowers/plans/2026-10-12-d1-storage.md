# D1 Storage Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Everything the owner does in BoeesTrack is stored in Cloudflare D1 and survives a reload; the live app starts empty.

**Architecture:** Four D1 tables (`entries`, `pots`, `categories`, `plan`) behind small Hono routes in the Worker, all under the existing Cloudflare Access check. The React app loads everything once on start, then applies each change on screen first and sends it in the background, undoing it with a short notice if the save fails.

**Tech Stack:** Cloudflare D1, Hono 4.11, React 19, TypeScript, Vitest (plain Node), wrangler 4.136.

**Spec:** `docs/superpowers/specs/2026-10-12-d1-storage-design.md`

## Global Constraints

- Style is tabs, double quotes, semicolons. No prettier, no formatter.
- Amounts are integer satang (฿1 = 100). Dates are stored as UTC milliseconds.
- Keep `compatibility_date` at `2026-08-18` in `wrangler.json`. If local D1 does not run on that date, report it; do not change the date.
- `ACCESS_BYPASS=true` goes only in the git-ignored `.dev.vars`, never in `wrangler.json`.
- Never fabricate real-looking transaction data. Never read, quote or commit anything from `slipsScreenShots/`.
- Design rules for any UI: tokens from `src/react-app/styles/tokens.css` only (no inline colours or font names), one red only, works at 320px with no horizontal overflow, tap targets of at least 44px (`var(--tap)`).
- Commit messages carry NO `Co-Authored-By` or "Generated with" lines (owner rule). Commit as the configured git user. Never push, never deploy, never merge to main.
- Keep shell commands simple (one command per call); do not `cd` out of the worktree.
- `.vibe-wise/` and `.gitignore` are not edited by this plan.

## Review Focus

1. A duplicate slip ref (same slip confirmed twice) must return 409, and the app must take the entries back with a notice, not leave a ghost entry.
2. A batch with a repeated entry id, or an entry id that already exists (double-tap on Save reuses `entry-${Date.now()}`), must be rejected without half-saving the batch.
3. Renaming a pot to a name that already exists must fail as one unit (409): the pot and its entries keep their old name, and the app undoes the rename.
4. Bad input (non-JSON body, float `at`, negative `spent`, `saved` with no pot, 201 entries) must return 400 and write nothing.
5. The app with completely empty data (no entries, no pots, ฿0 plan) must not crash or show NaN on Home, Insight, History and Savings, at 320px.
6. A pot whose name has spaces or Thai letters must rename correctly (the name travels in the URL path, so it must be URL-encoded).

```ts
// What travels between the app and the Worker, and what D1 rows look like.
// The Worker never trusts the browser: every body is checked here before it touches the database.

export const KINDS = ["spent", "received", "saved", "adjust"] as const;
export type WireKind = (typeof KINDS)[number];

// `at` is UTC milliseconds on the wire; only the app turns it into a Date.
export type WireEntry = {
	id: string;
	kind: WireKind;
	satang: number;
	at: number;
	category: string | null;
	note: string;
	slipRef?: string;
	pot?: string;
};
export type WirePot = { name: string; target: number; monthly: number };
export type WirePlan = { monthly: number; daily: "auto" | "off" | { fixed: number } };

export type Parsed<T> = { ok: true; value: T } | { ok: false; error: string };
const fail = (error: string): { ok: false; error: string } => ({ ok: false, error });

export const MAX_BATCH = 200;

const isObject = (v: unknown): v is Record<string, unknown> => typeof v === "object" && v !== null && !Array.isArray(v);
const isInt = (v: unknown): v is number => typeof v === "number" && Number.isSafeInteger(v);
const isText = (v: unknown, max: number): v is string => typeof v === "string" && v.trim() !== "" && v.length <= max;

function parseEntry(raw: unknown): Parsed<WireEntry> {
	if (!isObject(raw)) return fail("entry must be an object");
	const { id, kind, satang, at, category, note, slipRef, pot } = raw;
	if (!isText(id, 100)) return fail("bad id");
	if (!KINDS.includes(kind as WireKind)) return fail("bad kind");
	if (!isInt(satang) || satang === 0 || (kind !== "adjust" && satang < 0)) return fail("bad satang");
	if (!isInt(at) || at < 0) return fail("bad at");
	if (category !== null && !isText(category, 50)) return fail("bad category");
	if (typeof note !== "string" || note.length > 500) return fail("bad note");
	if (slipRef !== undefined && !isText(slipRef, 100)) return fail("bad slipRef");
	if (pot !== undefined && !isText(pot, 50)) return fail("bad pot");
	if (kind === "saved" && pot === undefined) return fail("a saved entry needs a pot");
	return {
		ok: true,
		value: {
			id,
			kind: kind as WireKind,
			satang,
			at,
			category: category as string | null,
			note,
			...(slipRef !== undefined ? { slipRef } : {}),
			...(pot !== undefined ? { pot } : {}),
		},
	};
}

export function parseEntries(body: unknown): Parsed<WireEntry[]> {
	if (!Array.isArray(body) || body.length === 0 || body.length > MAX_BATCH) return fail("expected 1 to 200 entries");
	const out: WireEntry[] = [];
	for (const raw of body) {
		const r = parseEntry(raw);
		if (!r.ok) return r;
		out.push(r.value);
	}
	if (new Set(out.map((e) => e.id)).size !== out.length) return fail("repeated id");
	return { ok: true, value: out };
}

export function parsePot(body: unknown): Parsed<WirePot> {
	if (!isObject(body)) return fail("pot must be an object");
	const { name, target, monthly } = body;
	if (!isText(name, 50)) return fail("bad name");
	if (!isInt(target) || target < 0) return fail("bad target");
	if (!isInt(monthly) || monthly < 0) return fail("bad monthly");
	return { ok: true, value: { name: name.trim(), target, monthly } };
}

export function parsePlan(body: unknown): Parsed<WirePlan> {
	if (!isObject(body)) return fail("plan must be an object");
	const { monthly, daily } = body;
	if (!isInt(monthly) || monthly < 0) return fail("bad monthly");
	if (daily === "auto" || daily === "off") return { ok: true, value: { monthly, daily } };
	if (isObject(daily) && isInt(daily.fixed) && daily.fixed >= 0) return { ok: true, value: { monthly, daily: { fixed: daily.fixed } } };
	return fail("bad daily");
}

export function parseCategory(body: unknown): Parsed<string> {
	if (!isObject(body) || !isText(body.name, 50)) return fail("bad name");
	return { ok: true, value: (body.name as string).trim() };
}

export type EntryRow = {
	id: string;
	kind: WireKind;
	satang: number;
	at: number;
	category: string | null;
	note: string;
	slip_ref: string | null;
	pot: string | null;
};

export function entryFromRow(r: EntryRow): WireEntry {
	return {
		id: r.id,
		kind: r.kind,
		satang: r.satang,
		at: r.at,
		category: r.category,
		note: r.note,
		...(r.slip_ref ? { slipRef: r.slip_ref } : {}),
		...(r.pot ? { pot: r.pot } : {}),
	};
}

export type PlanRow = { monthly: number; daily_mode: "auto" | "off" | "fixed"; daily_fixed: number | null };

export function planFromRow(r: PlanRow): WirePlan {
	return { monthly: r.monthly, daily: r.daily_mode === "fixed" ? { fixed: r.daily_fixed ?? 0 } : r.daily_mode };
}

export function planToColumns(plan: WirePlan): { mode: "auto" | "off" | "fixed"; fixed: number | null } {
	return typeof plan.daily === "string" ? { mode: plan.daily, fixed: null } : { mode: "fixed", fixed: plan.daily.fixed };
}
```

**Interfaces:**

- Produces (used by Task 4), all under `/api`:
  - `GET /data` → `{ entries: WireEntry[]; pots: WirePot[]; categories: string[]; plan: WirePlan }`
  - `POST /entries` body `WireEntry[]` → 201 `{ ok: true }`
  - `POST /pots` body `WirePot` → 201
  - `PUT /pots/:name` body `WirePot` → 200, 404 if no such pot
  - `PUT /plan` body `WirePlan` → 200
  - `POST /categories` body `{ name }` → 201
  - 400 with a short text on invalid input, 409 on a UNIQUE conflict.

- [ ] **Step 1: Write the routes**

Create `src/worker/data.ts`:

```ts
import { type Context, Hono } from "hono";
import {
	type EntryRow,
	type PlanRow,
	type WirePot,
	entryFromRow,
	parseCategory,
	parseEntries,
	parsePlan,
	parsePot,
	planFromRow,
	planToColumns,
} from "./wire";

export const data = new Hono<{ Bindings: Env }>();

// A UNIQUE clash (same entry id, slip ref, pot or category name) is the caller's mistake, not a crash.
async function guard(c: Context, run: () => Promise<Response>): Promise<Response> {
	try {
		return await run();
	} catch (e) {
		if (e instanceof Error && e.message.includes("UNIQUE")) return c.text("Already exists", 409);
		throw e;
	}
}

const json = (c: Context) => c.req.json().catch(() => null);

data.get("/data", async (c) => {
	const db = c.env.DB;
	const [entries, pots, categories, plan] = await db.batch([
		db.prepare("SELECT id, kind, satang, at, category, note, slip_ref, pot FROM entries ORDER BY at, id"),
		db.prepare("SELECT name, target, monthly FROM pots ORDER BY rowid"),
		db.prepare("SELECT name FROM categories ORDER BY position"),
		db.prepare("SELECT monthly, daily_mode, daily_fixed FROM plan WHERE id = 1"),
	]);
	return c.json({
		entries: (entries.results as EntryRow[]).map(entryFromRow),
		pots: pots.results as WirePot[],
		categories: (categories.results as { name: string }[]).map((r) => r.name),
		plan: planFromRow((plan.results as PlanRow[])[0]),
	});
});

data.post("/entries", async (c) => {
	const parsed = parseEntries(await json(c));
	if (!parsed.ok) return c.text(parsed.error, 400);
	return guard(c, async () => {
		const db = c.env.DB;
		const insert = db.prepare("INSERT INTO entries (id, kind, satang, at, category, note, slip_ref, pot) VALUES (?, ?, ?, ?, ?, ?, ?, ?)");
		await db.batch(parsed.value.map((e) => insert.bind(e.id, e.kind, e.satang, e.at, e.category, e.note, e.slipRef ?? null, e.pot ?? null)));
		return c.json({ ok: true }, 201);
	});
});

data.post("/pots", async (c) => {
	const parsed = parsePot(await json(c));
	if (!parsed.ok) return c.text(parsed.error, 400);
	return guard(c, async () => {
		const { name, target, monthly } = parsed.value;
		await c.env.DB.prepare("INSERT INTO pots (name, target, monthly) VALUES (?, ?, ?)").bind(name, target, monthly).run();
		return c.json({ ok: true }, 201);
	});
});

// The pot and its entries change in one batch, so a rename never leaves entries pointing at a pot that is gone.
data.put("/pots/:name", async (c) => {
	const parsed = parsePot(await json(c));
	if (!parsed.ok) return c.text(parsed.error, 400);
	return guard(c, async () => {
		const old = c.req.param("name");
		const { name, target, monthly } = parsed.value;
		const db = c.env.DB;
		const [pot] = await db.batch([
			db.prepare("UPDATE pots SET name = ?, target = ?, monthly = ? WHERE name = ?").bind(name, target, monthly, old),
			db.prepare("UPDATE entries SET pot = ? WHERE pot = ?").bind(name, old),
		]);
		if (pot.meta.changes === 0) return c.text("No such pot", 404);
		return c.json({ ok: true });
	});
});

data.put("/plan", async (c) => {
	const parsed = parsePlan(await json(c));
	if (!parsed.ok) return c.text(parsed.error, 400);
	const { mode, fixed } = planToColumns(parsed.value);
	await c.env.DB.prepare("UPDATE plan SET monthly = ?, daily_mode = ?, daily_fixed = ? WHERE id = 1").bind(parsed.value.monthly, mode, fixed).run();
	return c.json({ ok: true });
});

data.post("/categories", async (c) => {
	const parsed = parseCategory(await json(c));
	if (!parsed.ok) return c.text(parsed.error, 400);
	return guard(c, async () => {
		await c.env.DB.prepare("INSERT INTO categories (name, position) VALUES (?, (SELECT COALESCE(MAX(position), 0) + 1 FROM categories))")
			.bind(parsed.value)
			.run();
		return c.json({ ok: true }, 201);
	});
});
```

```ts
// A change is already on screen; the save runs behind it. If the save fails, take the change back and say so.
export async function saveOrUndo(save: () => Promise<unknown>, undo: () => void, onFail: (message: string) => void): Promise<void> {
	try {
		await save();
	} catch {
		undo();
		onFail("Couldn't save that, so it was taken back.");
	}
}
```

Create `src/react-app/lib/api.ts`:

```ts
import type { WireEntry, WirePlan, WirePot } from "../../worker/wire";
import type { BudgetPlan, Entry, Pot } from "./types";

// Dates travel as UTC milliseconds; this file is the only place they are converted.
export const toWire = (e: Entry): WireEntry => ({
	id: e.id,
	kind: e.kind,
	satang: e.satang,
	at: e.at.getTime(),
	category: e.category,
	note: e.note,
	...(e.slipRef ? { slipRef: e.slipRef } : {}),
	...(e.pot ? { pot: e.pot } : {}),
});

export const fromWire = (w: WireEntry): Entry => ({ ...w, at: new Date(w.at) });

async function call(method: string, path: string, body?: unknown): Promise<Response> {
	const res = await fetch(`/api${path}`, {
		method,
		...(body === undefined ? {} : { headers: { "content-type": "application/json" }, body: JSON.stringify(body) }),
	});
	if (!res.ok) throw new Error(`${method} ${path} failed: ${res.status}`);
	return res;
}

export async function loadAll(): Promise<{ entries: Entry[]; pots: Pot[]; categories: string[]; plan: BudgetPlan }> {
	const all = (await (await call("GET", "/data")).json()) as { entries: WireEntry[]; pots: WirePot[]; categories: string[]; plan: WirePlan };
	return { ...all, entries: all.entries.map(fromWire) };
}

export const addEntries = (entries: Entry[]) => call("POST", "/entries", entries.map(toWire)).then(() => undefined);
export const addPot = (pot: Pot) => call("POST", "/pots", pot).then(() => undefined);
export const editPot = (oldName: string, pot: Pot) => call("PUT", `/pots/${encodeURIComponent(oldName)}`, pot).then(() => undefined);
export const savePlan = (plan: BudgetPlan) => call("PUT", "/plan", plan).then(() => undefined);
export const addCategory = (name: string) => call("POST", "/categories", { name }).then(() => undefined);
```
